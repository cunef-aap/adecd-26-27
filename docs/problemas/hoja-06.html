Ejercicio 8 Este fragmento pretende medir los tres términos de Ecuación 6.6 sobre el experimento simulado del capítulo. Tiene tres errores conceptuales. Identifícalos y explica qué consecuencia tiene cada uno sobre lo que el fragmento imprime y sobre la conclusión que saca.

import torch

torch.set_default_dtype(torch.float64)

NOISE = 0.2
rejilla = torch.linspace(0, 2, 400)


def senal(x):
    return torch.sin(torch.pi * x)


def caracteristicas(x, grado):
    return torch.stack([x ** j for j in range(grado + 1)], dim=1)


def muchos_ajustes(grado, n=36, repeticiones=2000, semilla=0):
    torch.manual_seed(semilla)
    predicciones = torch.empty(repeticiones, len(rejilla))
    for r in range(repeticiones):
        xr = torch.rand(n) * 2
        yr = senal(xr) + NOISE * torch.randn(n)
        w = torch.linalg.lstsq(caracteristicas(xr, grado), yr).solution
        predicciones[r] = caracteristicas(rejilla, grado) @ w
    return predicciones


def descompone(grado, n=36):
    P = muchos_ajustes(grado, n=n)
    sesgo2 = ((P - senal(rejilla)) ** 2).mean().item()
    varianza = P[0].var(unbiased=False).item()
    return sesgo2, varianza, NOISE ** 2 + sesgo2 + varianza


for grado in (1, 3, 9):
    sesgo2, varianza, total = descompone(grado)
    print(f"grado {grado}: sesgo^2 {sesgo2:.4f}  varianza {varianza:.4f}"
          f"  riesgo esperado {total:.4f}")

print("el sesgo^2 del grado 1 baja a "
      f"{descompone(1, n=3000)[0]:.4f} con n = 3000, asi que el sesgo "
      "de la recta se corrige con mas datos")

Ejercicio 9 Este fragmento pretende elegir el grado del polinomio en los datos de Madrid y comparar el grado elegido con el modelo que entregó el capítulo 5. Recortador, codificador, TAMANO, RESTO y CATEGORICAS son los del capítulo, y cargar_datos devuelve el reparto por anfitrión. Tiene tres errores conceptuales. Identifícalos y explica qué consecuencia tiene cada uno.

import numpy as np
from sklearn.compose import ColumnTransformer
from sklearn.impute import SimpleImputer
from sklearn.linear_model import LinearRegression
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import PolynomialFeatures, StandardScaler

train_a, test_a, y_train, y_test = cargar_datos()


def pipeline_polinomio(grado):
    tamano = [("recortar", Recortador()),
              ("imputar", SimpleImputer(strategy="median",
                                        add_indicator=True)),
              ("expandir", PolynomialFeatures(grado,
                                              include_bias=False)),
              ("escalar", StandardScaler())]
    otras = [("recortar", Recortador()),
             ("imputar", SimpleImputer(strategy="median",
                                       add_indicator=True)),
             ("escalar", StandardScaler())]
    preparar = ColumnTransformer([
        ("tam", Pipeline(tamano), TAMANO),
        ("otras", Pipeline(otras), RESTO),
        ("cat", codificador(), CATEGORICAS)])
    return Pipeline([("preparar", preparar),
                     ("modelo", LinearRegression())])


perdidas = {}
for grado in (1, 2, 3, 4):
    modelo = pipeline_polinomio(grado).fit(train_a, y_train)
    perdidas[grado] = (modelo.predict(test_a) - y_test) ** 2
    print(f"grado {grado}: riesgo en test {perdidas[grado].mean():.1f}")

elegido = min(perdidas, key=lambda g: perdidas[g].mean())
print(f"grado elegido: {elegido}")

for grado in (elegido, 1):
    p = perdidas[grado]
    ee = p.std(ddof=1) / np.sqrt(len(p))
    print(f"grado {grado}: {p.mean():.1f} ± {ee:.1f}")
print("los dos intervalos se solapan, asi que la mejora sobre el "
      "capitulo 5 no se distingue del ruido")
